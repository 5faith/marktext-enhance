vite.config.js 核心结构
import { defineConfig } from 'vite'
import vue from '@vitejs/plugin-vue'
import electron from 'vite-plugin-electron'
export default defineConfig({
plugins: [vue(), electron({...})],
resolve: { alias: {/* 保持现有别名 */} },
root: 'src/', // 项目根目录
})
